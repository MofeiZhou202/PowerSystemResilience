\begin{align}
P_i &= P_i^0 \bigl(z_P |V_i|^2 + i_P |V_i| + p_P\bigr), \\

\end{align}
Default weights $\{1, 0, 0\}$ recover the constant-power (PQ) model.

\subsection{Fischer-Burmeister NCP Functions}
\label{sec:ncp}
% ------------------------------------------------------------

\texttt{include/hacdcpf/power\_flow/ncp\_functions.hpp} provides the
complementarity functions used by the semi-smooth Newton PV/PQ handling.

\subsubsection{Standard Fischer-Burmeister Function}

The smoothed Fischer-Burmeister (FB) NCP function is:
\begin{equation}
  \mathrm{FB}(a, b) = \sqrt{a^2 + b^2 + \varepsilon} - a - b, \quad \varepsilon = 10^{-14}.
\end{equation}
$\mathrm{FB}(a, b) = 0 \iff a \geq 0,\; b \geq 0,\; ab = 0$.

Its partial derivatives are:
\begin{equation}
  \frac{\partial \mathrm{FB}}{\partial a} = \frac{a}{\sqrt{a^2+b^2+\varepsilon}} - 1, \qquad
  \frac{\partial \mathrm{FB}}{\partial b} = \frac{b}{\sqrt{a^2+b^2+\varepsilon}} - 1.
\end{equation}

\subsubsection{PV/PQ Complementarity Equation}

For a generator bus with reactive limits $[Q_{min}, Q_{max}]$ and
voltage setpoint $\bar{V}$:
\begin{equation}
  F_{NCP} = \mathrm{FB}(Q_{max} - Q_g,\; V - \bar{V}) +
             \mathrm{FB}(Q_g - Q_{min},\; \bar{V} - V).
\end{equation}
$F_{NCP}=0$ enforces three complementary conditions simultaneously:
\begin{itemize}
  \item $Q_{min} < Q_g < Q_{max}$: voltage regulation $V = \bar{V}$ (PV).
  \item $Q_g = Q_{max}$: $V \leq \bar{V}$ (reactive limit, switch to PQ).
  \item $Q_g = Q_{min}$: $V \geq \bar{V}$ (lower limit, switch to PQ).
\end{itemize}

Jacobian w.r.t.\ $(Q_g, V)$:
\begin{equation}
  \frac{\partial F_{NCP}}{\partial Q_g} = -\partial_a\mathrm{FB}_1 + \partial_a\mathrm{FB}_2, \qquad
  \frac{\partial F_{NCP}}{\partial V} = \partial_b\mathrm{FB}_1 - \partial_b\mathrm{FB}_2.
\end{equation}

\subsubsection{Smooth ($\mu$-Perturbed) Fischer-Burmeister}

Phase~2 of \texttt{RobustNonlinearOptions} uses a $\mu$-perturbed smooth FB:
\begin{equation}
  \varphi_\mu(a, b) = \sqrt{a^2 + b^2 + 2\mu} - a - b, \quad \mu > 0.
\end{equation}
As $\mu \to 0$, this recovers the standard near-zero FB.  $\mu$ is annealed
from \texttt{ncp\_mu0} to \texttt{ncp\_mu\_min} as the residual decreases,
using a two-phase schedule:
\begin{equation}
  \mu_{k+1} = \begin{cases}
    \texttt{ncp\_mu\_factor\_coarse} \cdot \mu_k & \text{if } \|F\| > \theta \cdot \|F_0\|, \\
    \texttt{ncp\_mu\_factor} \cdot \mu_k & \text{otherwise (fine phase)}.
  \end{cases}
\end{equation}
where $\theta = \texttt{ncp\_mu\_phase\_transition}$ (default 0.1).

The smooth PV/PQ NCP residual and its Jacobian follow the same structure
as the standard version but with \texttt{smooth\_fb} replacing
\texttt{fischer\_burmeister}.

Activity hysteresis (\texttt{enable\_activity\_hysteresis=true}) prevents
PV/PQ chattering by requiring \texttt{min\_active\_set\_hold\_iters}~$=3$
iterations to elapse between consecutive active-set changes.
