\subsection{结果结构摘要}
\begin{itemize}
  \item \fld{BusFaultResult}（:DCBreakerDutyResult）/\fld{SCResult}（:DCFaultResult）：概览 $Z_{thevenin}$、$I_{fault}$、
        \fld{sk\_mva}、\fld{ikpp\_ka}；
  \item \fld{SCDetailedResult}（:dc\_bus\_fault\_level）：\fld{solved}、逐母线 \fld{SCDetailedBusResult}（:114，
        $I_k''/i_p/I_b/I_k/I_{th}$ 及分源贡献）、逐支路 \fld{SCDetailedBranchResult}（:dc\_bus\_fault\_level）、
        换流器贡献 \fld{SCConverterContributionResult}（:dc\_bus\_fault\_level）；
  \item \fld{DCFaultResult}：\fld{i\_fault\_ka}、\fld{r\_thevenin\_pu}、\fld{breaker\_duties}
        （\fld{DCBreakerDutyResult}）。
\end{itemize}
